"use strict";
const $ = (id) => document.getElementById(id);
let csrf = sessionStorage.getItem("manas-owner-csrf") || "", screen = "overview", generation = 0;
const labels = (value) => value.split(",").map((item) => item.trim()).filter(Boolean);
const key = () => crypto.randomUUID();
const el = (tag, text, cls) => { const node = document.createElement(tag); if (text !== undefined) node.textContent = text; if (cls) node.className = cls; return node; };
function message(text, error = false) { $("message").textContent = text; $("message").className = error ? "error" : ""; }
async function api(path, input = {}) {
 const response = await fetch("/api/" + path, { method: "POST", headers: { "Content-Type": "application/json", "X-Manas-Csrf": csrf }, body: JSON.stringify(input), credentials: "same-origin" });
 const result = await response.json();
 if (!result.ok) { if (response.status === 401) { csrf = ""; sessionStorage.removeItem("manas-owner-csrf"); $("connection").textContent = "Owner sign-in required"; } throw new Error(result.error.message + " " + result.error.suggestedFix); }
 return result.data;
}
const op = (operation, input = {}) => api("operation", { operation, input });
function badge(text) { return el("span", text, "badge " + String(text).replace(/[^a-z]/g, "")); }
function detail(parent, value, title = "Inspect evidence and receipt") { const node = el("details"); node.append(el("summary", title), el("pre", JSON.stringify(value, null, 2))); parent.append(node); }
function card(title, description) { const node = el("article", undefined, "card"); node.append(el("h2", title)); if (description) node.append(el("p", description, "muted")); return node; }
function button(text, action, primary = false) {
 const node = el("button", text, primary ? "primary" : "");
 node.type = "button";
 node.addEventListener("click", async () => { node.disabled = true; try { await action(); } catch (error) { message(error.message, true); } finally { node.disabled = false; } });
 return node;
}
function actions(parent, ...nodes) { const row = el("div", undefined, "actions"); row.append(...nodes); parent.append(row); }
function field(form, name, label, type = "text", value = "", required = true) {
 const id = "field-" + key(), wrapper = el("div"), title = el("label", label); title.htmlFor = id;
 const input = el(type === "textarea" ? "textarea" : "input"); if (type !== "textarea") input.type = type;
 input.id = id; input.name = name; input.value = value; input.required = required;
 wrapper.append(title, input); form.append(wrapper); return input;
}
function choice(form, name, label, options) {
 const wrapper = el("div"), title = el("label", label), select = el("select"), id = "field-" + key(); select.id = id; select.name = name; title.htmlFor = id;
 for (const [value, text] of options) { const option = el("option", text); option.value = value; select.append(option); }
 wrapper.append(title, select); form.append(wrapper); return select;
}
function form(parent, title, submitLabel, configure, submit) {
 const section = card(title), node = el("form"); configure(node); const send = el("button", submitLabel, "primary"); send.type = "submit"; const row = el("div", undefined, "actions"); row.append(send); node.append(row);
 let fingerprint = "", requestKey = key();
 node.addEventListener("submit", async (event) => { event.preventDefault(); send.disabled = true; message("Working…"); try { const values = Object.fromEntries(new FormData(node)), nextFingerprint = JSON.stringify(values); if (fingerprint !== nextFingerprint) { fingerprint = nextFingerprint; requestKey = key(); } await submit(values, requestKey); } catch (error) { message(error.message, true); } finally { send.disabled = false; } }); section.append(node); parent.append(section); return section;
}
async function confirmChange(title, text) {
 $("confirm-title").textContent = title; $("confirm-description").textContent = text; const dialog = $("confirm-dialog"); dialog.returnValue = ""; dialog.showModal();
 return new Promise((resolve) => dialog.addEventListener("close", () => resolve(dialog.returnValue === "confirm"), { once: true }));
}
function empty(parent, text) { parent.append(card("Nothing to show", text)); }
function evidence(parent, items) {
 for (const item of items || []) { const quote = el("blockquote", item.text || "", "excerpt"); parent.append(quote); detail(parent, item.citation || item, "Source citation"); }
}

async function load(next = screen) {
 screen = next; const request = ++generation, content = $("content"); content.replaceChildren(); message("Loading…");
 $("heading").textContent = screens[screen][0]; $("description").textContent = screens[screen][1];
 for (const node of document.querySelectorAll("[data-screen]")) node.setAttribute("aria-current", node.dataset.screen === screen ? "page" : "false");
 try { await render[screen](content, () => request === generation); if (request === generation) message(""); }
 catch (error) { if (request === generation) message(error.message, true); }
}
